1"""Issue classifier — two-stage: rules first, then LLM fallback. 

2 

3Stage 1 (always): keyword matching on title/body/labels. 

4Stage 2 (optional): LLM call when rules are uncertain (UNKNOWN or confidence ≤ 0.6). 

5 

6Callers use the sync ``classify()`` for pure rules, or ``async_classify()`` 

7to get LLM-enhanced results when available. 

8 

9==================================================================== 

10Issue 分类器 —— 两阶段分类：规则匹配 + LLM 兜底 

11==================================================================== 

12 

13阶段 1（规则匹配，始终执行）： 

14 基于关键词词典对 Issue 的标题、正文、标签进行匹配打分。 

15 返回置信度最高的类别及信号列表。 

16 

17阶段 2（LLM 兜底，可选）： 

18 当 LLM API 可用时，对规则分类结果置信度低的情况， 

19 调用 LLM 进行二次判断，以提高分类准确度。 

20 

21分类类别（IssueCategory）： 

22 - BUG 缺陷/崩溃报告 

23 - FEATURE_REQUEST 功能建议/新需求 

24 - QUESTION 使用咨询/疑问 

25 - DOCUMENTATION 文档问题 

26 - DUPLICATE 重复 Issue 

27 - INFO_NEEDED 信息不足，需要补充 

28 - INVALID 无效/不在范围内 

29 - MAINTENANCE 维护/重构/依赖更新 

30 - UNKNOWN 无法归类（需人工处理） 

31 

32评分规则： 

33 - 标签中的关键词命中权重 ×2（标签通常比正文标题更可信）。 

34 - 正文为空自动增加 INFO_NEEDED 信号。 

35 - 置信度 = 最高得分 / 总得分，限制在 [0.35, 0.95] 区间内。 

36 - 仅由空正文触发的信号置信度强制为 0.3（证据不足，需 LLM 二次判断）。 

37 

38使用方式： 

39 classifier = IssueClassifier() 

40 # 同步：纯规则（快速，无需 LLM） 

41 result = classifier.classify(title="Bug: crash", body="...", labels=["bug"]) 

42 # 异步：规则 + LLM（高置信度时跳过 LLM，低置信度时调用 LLM 增强） 

43 result = await classifier.async_classify(title="...", body="...", labels=[]) 

44"""

89class IssueClassifier: 

90 """Classify an issue into an ``IssueCategory``. 

91 

92 两阶段分类器： 

93 - classify() 纯规则，同步、零依赖。 

94 - async_classify() 规则 + LLM 兜底，异步，需要 LLM API。 

95 

96 Usage:: 

97 

98 # Sync: pure rule-based (fast, no LLM needed) 

99 result = classifier.classify(title="Bug: crash", body="...", labels=["bug"]) 

100 

101 # Async: rules + LLM fallback for uncertain cases 

102 result = await classifier.async_classify(title="...", body="...", labels=[]) 

103 """

105 # ── Sync: rule-based classification ─────────────────────────────── 

106 # 同步规则分类 

107 

108 def classify(self, title: str, body: str | None, labels: list[str]) -> IssueClassification: 

109 """Score each category by keyword matching and return the top result. 

110 

111 - Keyword matches in labels score 2×. 

112 - Empty body adds an ``INFO_NEEDED`` signal. 

113 - Confidence is ``top_score / total_score``, clamped to ``[0.35, 0.95]``. 

114 

115 步骤： 

116 1. 将标题、正文、标签拼接为统一文本 → 小写化。 

117 2. 单独提取标签文本（用于 2× 加权检测）。 

118 3. 遍历关键词词典，对每个命中累加分数： 

119 - 标签中命中：+2 分（高权重，标签通常更可靠）。 

120 - 标题/正文中命中：+1 分。 

121 4. 正文为空 → 追加 INFO_NEEDED 信号（权重较低）。 

122 5. 计算置信度 → 返回得分最高的类别。 

123 

124 Args: 

125 title: Issue 标题。 

126 body: Issue 正文（可为 None 表示空）。 

127 labels: 标签列表（如 ``["bug", "triaged"]``）。 

128 

129 Returns: 

130 IssueClassification 包含类别、置信度、推理原因等字段。 

131 """ 

132 text = " ".join([title, body or "", " ".join(labels)]).lower() 

133 label_text = " ".join(labels).lower() 

134 scores: Counter[IssueCategory] = Counter() 

135 signals: list[str] = [] 

136 

137 for category, keywords in KEYWORDS.items(): 

138 for keyword in keywords: 

139 if keyword in text: 

140 # 标签命中权重 ×2，正文/标题命中权重 ×1 

141 score = 2 if keyword in label_text else 1 

142 scores[category] += score 

143 signals.append(f"{category.value}:{keyword}") 

144 

145 # 空正文提升：没有正文的 issue 大概率信息不足。 

146 # 但仅当这是唯一信号时，置信度保持较低以便 LLM 兜底有分析机会。 

147 empty_body_signals: list[str] = [] 

148 if not (body and body.strip()) and scores[IssueCategory.INFO_NEEDED] == 0: 

149 scores[IssueCategory.INFO_NEEDED] += 1 

150 empty_body_signals = ["info_needed:empty_body"] 

151 

152 if not scores: 

153 # 完全没有任何关键词命中 → 返回 UNKNOWN 

154 return IssueClassification( 

155 category=IssueCategory.UNKNOWN, 

156 confidence=0.2, 

157 reason="No strong label, title, or body keyword matched the rule classifier.", 

158 suggested_action=SUGGESTED_ACTIONS[IssueCategory.UNKNOWN], 

159 signals=[], 

160 ) 

161 

162 # 取最高分作为分类结果 

163 category, score = scores.most_common(1)[0] 

164 total = sum(scores.values()) 

165 # 置信度 = 最高分占总分的比例，限制在 [0.35, 0.95] 

166 confidence = min(0.95, max(0.35, score / total)) 

167 # 如果空正文启发式是唯一信号，置信度应为 0.3，便于 LLM 兜底 

168 if empty_body_signals and len(scores) == 1: 

169 confidence = 0.3 

170 signals.extend(empty_body_signals) 

171 return IssueClassification( 

172 category=category, 

173 confidence=round(confidence, 2), 

174 reason=f"Matched {score} signal(s) for {category.value}.", 

175 suggested_action=SUGGESTED_ACTIONS[category], 

176 signals=signals[:8], # 最多保留 8 个信号 

177 )

197 body: str | None,

208 async def _llm_classify( 

209 self, 

210 title: str, 

211 body: str | None, 

212 labels: list[str], 

213 ) -> IssueClassification | None: 

214 """Ask the LLM to classify the issue. Returns ``None`` on failure. 

215 

216 LLM 分类调用： 

217 

218 构造结构化的 Prompt → 调用 LLM → 解析 JSON 返回结果。 

219 

220 返回 None 的情况： 

221 - LLM API 调用失败（网络问题、API 错误等）。 

222 - LLM 返回的 JSON 无法解析。 

223 - LLM 返回的类别字符串不是有效的 IssueCategory。 

224 

225 Args: 

226 title: Issue 标题。 

227 body: Issue 正文。 

228 labels: 标签列表。 

229 

230 Returns: 

231 IssueClassification 或 None（失败时）。 

232 """ 

233 categories_str = ", ".join(c.value for c in IssueCategory) 

234 prompt = ( 

235 f"Classify this GitHub issue into exactly one category: {categories_str}.\n\n" 

236 f"Title: {title}\n" 

237 f"Body: {body or '(empty)'}\n" 

238 f"Labels: {', '.join(labels) if labels else '(none)'}\n\n" 

239 "Respond in JSON only:\n" 

240 '{"category": "<category>", "confidence": 0.0-1.0, ' 

241 '"reason": "<brief reason, include duplicate issue# if applicable>", ' 

242 '"signals": ["<key evidence>"], ' 

243 '"auto_reply_draft": "(if this is a question, info_needed, duplicate, ' 

244 'or documentation issue, write a draft reply in Chinese; ' 

245 'for duplicate, mention which issue it duplicates); ' 

246 'otherwise leave empty"}\n' 

247 "confidence must be >= 0.5 if you are sure, < 0.5 if uncertain." 

248 ) 

249 

250 try: 

251 completion = await tracked_chat_completion( 

252 self._client, 

253 model=self._cfg.llm_model, 

254 messages=[ 

255 { 

256 "role": "system", 

257 "content": ( 

258 "You are an issue triage assistant for an open-source project. " 

259 "Classify GitHub issues with precision. " 

260 "For non-code issues (question, info_needed, duplicate, documentation), " 

261 "write a helpful auto-reply draft in Chinese. " 

262 "Respond in JSON only, no markdown." 

263 ), 

264 }, 

265 {"role": "user", "content": prompt}, 

266 ], 

267 temperature=0.3, # 低温度保证相对一致的分类结果 

268 max_tokens=500, # 限制输出长度避免浪费 token 

269 ) 

270 except Exception as exc: 

271 import logging 

272 logging.getLogger(__name__).warning("LLM classification failed: %s", exc) 

273 return None 

274 

275 raw = (completion.choices[0].message.content or "").strip() 

276 if not raw: 

277 return None 

278 

279 try: 

280 data = json.loads(raw) 

281 except json.JSONDecodeError: 

282 return None 

283 

284 # 验证并标准化类别的字符串 

285 category_str = (data.get("category") or "").strip().lower() 

286 try: 

287 category = IssueCategory(category_str) 

288 except ValueError: 

289 return None # 无效类别，返回 None 回退到规则 

290 

291 confidence = min(1.0, max(0.0, float(data.get("confidence", 0.5)))) 

292 signals = data.get("signals", []) 

293 if not isinstance(signals, list): 

294 signals = [] 

295 signals = [str(s) for s in signals[:8]] 

296 

297 auto_reply_draft = str(data["auto_reply_draft"]).strip() if data.get("auto_reply_draft") else None 

298 

299 return IssueClassification( 

300 category=category, 

301 confidence=round(confidence, 2), 

302 reason=data.get("reason", f"LLM classified as {category.value}.") or "", 

303 suggested_action=SUGGESTED_ACTIONS.get(category, ""), 

304 signals=signals, 

305 auto_reply_draft=auto_reply_draft, 

306 )

311 def summarize(self, categories: list[IssueCategory]) -> list[CategorySummary]: 

312 """Aggregate a list of categories into a sorted summary (most frequent first). 

313 

314 将多个分类结果聚合为频率统计摘要，按出现次数降序排列。 

315 

316 Args: 

317 categories: Issue 类别列表。 

318 

319 Returns: 

320 CategorySummary 列表（按出现次数降序）。 

321 """ 

322 counter = Counter(category.value for category in categories) 

323 return [CategorySummary(category=category, count=count) for category, count in counter.most_common()]
